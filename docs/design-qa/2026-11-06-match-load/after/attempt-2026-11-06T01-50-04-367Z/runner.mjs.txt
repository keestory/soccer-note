function freshMode(options = {}) {
  return { failTeams: false, failMemberships: false, failMatches: false, empty: false, deniedWrites: [], reads: [], holds: [], nextTeams: null, timezoneId: 'Asia/Seoul', browserLocale: 'ko-KR', ...options }
}
const nativeContext = browser.newContext.bind(browser)
browser.newContext = async options => {
  const own = mode
  const context = await nativeContext({ ...options, timezoneId: own.timezoneId, locale: own.browserLocale })
  const nativePage = context.newPage.bind(context)
  context.newPage = async () => {
    await context.route('**/*', async route => {
      const request = route.request(), url = new URL(request.url()), method = request.method()
      if (!['GET', 'HEAD', 'OPTIONS'].includes(method)) {
        own.deniedWrites.push({ method, path: url.pathname })
        return route.abort('blockedbyclient')
      }
      if (url.hostname !== 'soccer-note-test.supabase.co' || method !== 'GET' || !url.pathname.startsWith('/rest/v1/')) return route.fallback()
      const table = url.pathname.split('/').at(-1)
      const read = { table, teamId: url.searchParams.get('team_id'), userId: url.searchParams.get('user_id'), result: 'fixture' }
      own.reads.push(read)
      const respond = async (body, code = 200) => {
        read.result = code
        try { await route.fulfill({ status: code, contentType: 'application/json', body: JSON.stringify(body), headers: { 'access-control-allow-origin': '*' } }) }
        catch (error) { if (!/closed|handled|Interception|Fetch domain/.test(error.message)) throw error }
      }
      if (table === 'teams' && own.nextTeams) {
        const pending = own.nextTeams; own.nextTeams = null; own.holds.push(pending); pending.entered = true
        await pending.wait; pending.released = true
        if (pending.fail) return respond({ message: 'Synthetic held team-list failure', code: 'QA_FAIL' }, 500)
      }
      if ((table === 'teams' && own.failTeams) || (table === 'team_members' && own.failMemberships) || (table === 'matches' && own.failMatches)) return respond({ message: 'Synthetic lookup failure', code: 'QA_FAIL' }, 500)
      if (table === 'matches' && own.empty) return respond([])
      return route.fallback()
    })
    const page = await nativePage()
    page.setDefaultNavigationTimeout(30000)
    if (own.instant) await page.clock.setFixedTime(new Date(own.instant))
    return page
  }
  return context
}

async function shot(page, name) {
  await page.evaluate(() => document.fonts.ready)
  await page.screenshot({ path: `${runOutput}/${name}.png`, fullPage: false })
  const view = await page.evaluate(() => ({ width: innerWidth, height: innerHeight, dpr: devicePixelRatio, scrollWidth: document.documentElement.scrollWidth, lang: document.documentElement.lang, fontStatus: document.fonts.status }))
  report.screenshots.push({ file: `${attempt}/${name}.png`, ...view })
  return view
}
async function inspect(page) {
  return page.evaluate(() => ({ path: location.pathname, header: document.querySelector('header')?.innerText, main: document.querySelector('main')?.innerText, status: document.querySelector('[data-testid="matches-load-status"]')?.innerText ?? null, alerts: [...document.querySelectorAll('[role="alert"]')].map(x => x.textContent), matchRows: document.querySelectorAll('main a[href^="/match/"]').length, activeElement: document.activeElement?.tagName, activeText: document.activeElement?.textContent?.trim().slice(0, 120) }))
}
async function failed(page, cached = false) {
  await status(page).getByRole('alert').waitFor()
  assert.equal(await status(page).getByRole('alert').textContent(), failureText)
  assert.equal(await retry(page).getAttribute('aria-disabled'), 'false')
  if (cached) assert.match(await status(page).innerText(), /이전에 불러온 기록/)
  else {
    assert.equal(await page.getByText(emptyText, { exact: true }).count(), 0)
    assert.equal(await page.locator('main section[aria-label="팀 전적"]').count(), 0)
    assert.equal(await rows(page).count(), 0)
  }
}
async function ready(page, teamIndex = 0) {
  await page.getByText(matches[teamIndex].opponent, { exact: true }).waitFor()
  await until(async () => await status(page).count() === 0, 'load notice removed')
  assert((await page.locator('header').innerText()).includes(teams[teamIndex].name))
  assert.equal(await rows(page).count(), 1)
}
async function makeFixture(options = {}) {
  mode = freshMode(options.mode)
  const f = await fixture({ route: '/matches', width: options.width || 390, locale: options.locale || 'ko' })
  f.mode = mode
  return f
}
async function closeFixture(f) {
  for (const pending of f.mode.holds) pending.release()
  f.mode.nextTeams?.release()
  await f.close()
}
async function withFixture(work, options) {
  let f
  try {
    f = await makeFixture(options)
    const evidence = await work(f)
    assert.equal(f.state.writes.length, 0)
    assert.deepEqual(f.mode.deniedWrites, [])
    return { ...evidence, syntheticWrites: 0, productionWrites: 0, reads: f.mode.reads }
  } finally { if (f) await closeFixture(f) }
}
async function run(name, work) {
  if (filter && !name.includes(filter)) return
  const started = Date.now()
  try { report.cases.push({ name, status: 'pass', ...(await work()), durationMs: Date.now() - started }) }
  catch (error) { report.cases.push({ name, status: 'fail', error: error.stack, durationMs: Date.now() - started }) }
  console.log(JSON.stringify({ name, status: report.cases.at(-1).status, error: report.cases.at(-1).error }))
}
